# Function to fit model and create calibration plot
# ------------------------------------------------------------

make_calibration_plot <- function(data, outcome, plot_title) {

  # Keep complete cases for this outcome and the four CDI skills
  model_data <- data %>%
    select(
      pitcher_name,
      all_of(outcome),
      z_stuff,
      z_command,
      z_miss,
      z_contact
    ) %>%
    filter(complete.cases(.))

  # Fit regression model
  model_formula <- as.formula(
    paste(
      outcome,
      "~ z_stuff + z_command + z_miss + z_contact"
    )
  )

  model <- lm(
    formula = model_formula,
    data = model_data
  )

  # Add observed and predicted values
  calibration_data <- model_data %>%
    mutate(
      observed = .data[[outcome]],
      predicted = predict(model),
      residual = observed - predicted
    )

  # Create grouped calibration values
  # Each group contains approximately 10% of the sample
  calibration_bins <- calibration_data %>%
    mutate(
      prediction_bin = ntile(predicted, 10)
    ) %>%
    group_by(prediction_bin) %>%
    summarise(
      mean_predicted = mean(predicted, na.rm = TRUE),
      mean_observed = mean(observed, na.rm = TRUE),
      n = n(),
      .groups = "drop"
    )

  # Calculate metrics
  model_summary <- summary(model)

  rmse <- sqrt(mean(calibration_data$residual^2))
  mae <- mean(abs(calibration_data$residual))

  metrics <- tibble(
    outcome = outcome,
    n = nrow(calibration_data),
    adjusted_R2 = model_summary$adj.r.squared,
    RMSE = rmse,
    MAE = mae
  )

  # Axis limits shared by observed and predicted values
  axis_limits <- range(
    c(
      calibration_data$observed,
      calibration_data$predicted
    ),
    na.rm = TRUE
  )

  # Calibration plot
  plot <- ggplot(
    calibration_data,
    aes(x = predicted, y = observed)
  ) +
    geom_point(
      alpha = 0.45,
      size = 2
    ) +

    # Perfect calibration line
    geom_abline(
      intercept = 0,
      slope = 1,
      linetype = "dashed",
      linewidth = 1
    ) +

    # Actual fitted relationship
    geom_smooth(
      method = "lm",
      se = TRUE,
      linewidth = 1
    ) +

    # Mean observed and predicted values by decile
    geom_point(
      data = calibration_bins,
      aes(
        x = mean_predicted,
        y = mean_observed
      ),
      inherit.aes = FALSE,
      size = 3
    ) +

    geom_line(
      data = calibration_bins,
      aes(
        x = mean_predicted,
        y = mean_observed
      ),
      inherit.aes = FALSE,
      linewidth = 1
    ) +

    coord_equal(
      xlim = axis_limits,
      ylim = axis_limits
    ) +

    labs(
      title = plot_title,
      subtitle = paste0(
        "Adjusted R² = ",
        round(model_summary$adj.r.squared, 3),
        " | RMSE = ",
        round(rmse, 3),
        " | N = ",
        nrow(calibration_data)
      ),
      x = "Predicted value",
      y = "Observed value",
      caption = paste(
        "Dashed line = perfect calibration;",
        "connected points = observed and predicted means by decile."
      )
    ) +

    theme_minimal(base_size = 12) +
    theme(
      plot.title = element_text(face = "bold"),
      panel.grid.minor = element_blank()
    )

  return(
    list(
      model = model,
      data = calibration_data,
      bins = calibration_bins,
      metrics = metrics,
      plot = plot
    )
  )
}

The calibration results reinforce the rest of your validation. The 2025 models demonstrate both strong discrimination (high R²) and good calibration (predictions closely match observed values), while the 2026 models retain the expected relationship despite lower precision, providing evidence that the CDI framework generalizes beyond the season on which it was developed. This is exactly the pattern you would expect from a model trained on one season and evaluated on an independent, partially completed season, making it a strong external validation of the framework.
